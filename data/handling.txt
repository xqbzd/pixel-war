# 全局手感辅助（测试场里每台车都走这一套）。参数直接填，单位写在注释里。
#
# ★ "只在地面上才有效"：**有轮着地 = 满效果，一个都没着地 = 完全不生效**
#   （不按着地轮数的比例缩放 —— 压着两个轮子也算满）。
#   腾空↔落脚之间的过渡由下面 `ground_blend_seconds` 控制，落地是平滑恢复。
#
# 改完**进一次新场景**就生效（按文件修改时间重读，不用重启编辑器）。

# ① 下压力（模拟气动力）：力 = 系数 × **当前车重**，方向朝下，作用在车身上。
#    系数就是"**多少倍车重**"（无量纲、直接读得懂）：
#      0.3 = 温和；0.5 = 明显（推荐起点）；1.0 = 车"重了一倍"（很硬）。
#    **不随速度变**了（以前是 ×速度²）—— 要的是"一直更贴地"（压住悬挂 → 轮胎法向力
#    更大 → 抓地更大），不是"越快越贴"。
#    换算是这样：这台车 2270 kg（≈111 kN 重），写 0.5 = 一直有 56 kN 压下去。
#    你之前那套（×速度²、系数 10）在 10 m/s 时≈0.22 倍车重、20 m/s 时≈0.9 倍车重 ——
#    现在是个常数，按"你最常在哪个速度段开车"来定。
#    车重取**当前**值：零件被打掉、车变轻 → 下压力跟着变小。
#    0 = 关掉；想更贴地就加大（注意别把悬挂压穿，看 wheel_suspension_travel）。
downforce = 0.5

# ② 自动回正力矩：把车身往"竖直"方向拧回来，滚转和俯仰都管。
#    单位是**"车身倾到 90° 时给多大"**（N·m），实际按 sin(倾角) 缩放：
#      倾 30° → 力矩 × 0.5，倾 10° → × 0.17（所以是"越歪越拉，正了就松手"）。
#    参考：这台车靠重力自身的回正力矩，在倾 30° 时约 2270 kg ×19.6 格/秒² ×重心高×0.5。
#    （标定改成 1 格 = 0.5 米后重力是 19.6 格/秒² —— 比 0.2 米时弱了 2.5 倍，
#     所以同样写 60000，现在相对重力是**更硬**的。重心高按 1.5 格算大约 33000 N·m。）
#    写 60000 ≈ 在 30° 时补上重力的四成多；0 = 关掉。
#    ⚠️ 轮子文件里的"侧倾稳定器"（wheel_anti_roll_*）和它是**叠加**的，
#       两个一起开会很硬（像有根看不见的杆子）。建议只留一个。
upright_torque = 0

# 回正阻尼：相对临界阻尼（1.0 = 刚好不回摆，越小回正越"弹"）。0 = 不阻尼。
upright_damping = 0.6

# ③ 松手回正（偏航）：高速打方向再松开后，车身还会"自己转一会儿"才直 ——
#    这一条专门治它。**只有不给转向输入（方向键全松）时才生效**，所以打方向那段
#    的转向手感完全不受影响（别的地方也不用改）。
#    参数 = **时间常数**（秒）：车身"自转速度"按这个时间常数指数衰减，
#      0.35 → 0.35 秒掉到 37%、1 秒后基本不转了（推荐起点）
#      0.20 → 更"撒手就正"，更利落；0.80 → 明显更溜，能拖着尾巴走
#      0 = 关掉（回到只有轮胎自己那点阻尼的老样子）
#    和上面两条一样：只在地面上生效（有轮着地就算满）。
yaw_center_seconds = 0.15

# ④ 腾空↔落地的过渡时间（秒）：上面三条"生效强度"从 0 涨到 1（或反向）花多久。
#    有轮着地 → 目标满、一个没着地 → 目标 0，然后按这个时间线性过渡。
#    0.15 = 落下来 0.15 秒内恢复满效果（不会"啪"地跳一下）；写 0 = 立刻切换。
ground_blend_seconds = 0.15
