# 单位：距离用**格**（1 格 = 0.5 米），时间用**秒**，角度用**度**。
# 语法：`键 = 数值`，`#` 后面是注释；不认识的键、写歪的行会被忽略（保留默认值）。

# ── 距离 / 开火 ──────────────────────────────────────────────────────────────
# 到目标这个距离之前"冲过去"，到了就绕圈（260 格 = 130 米）
bot_standoff = 260
# 比这更近就倒车拉开（120 格 = 60 米）
bot_too_close = 120
# 开火距离上限（700 格 = 350 米）
bot_fire_range = 700
# 炮口方向和瞄准方向的点积阈值（0.997 ≈ 4.4°）—— 没对准就不浪费子弹
bot_aim_tolerance = 0.997
# 多少秒重新挑一次目标
bot_retarget_seconds = 2.0

# ── 驾驶 ────────────────────────────────────────────────────────────────────
# 给油但速度低于这个值（格/秒）就算"卡住"
bot_stuck_speed = 1.5
# 卡住多久（秒）开始倒车
bot_stuck_seconds = 1.5
# 倒车持续多久（秒）
bot_reverse_seconds = 1.2
# 前方障碍探测距离（格）
bot_probe = 42
# 绕障碍时偏转的角度（弧度，1.2 ≈ 69°）
bot_avoid_angle = 1.2

# ── 反应 / 精度（2026-10-18 新增）───────────────────────────────────────────
# 看到敌人之后要过多久（秒）才允许开火
bot_reaction_seconds = 0.3
# 刚看到敌人时，瞄点相对"核心"的最大随机偏差（度）
bot_aim_spread_deg = 2.0
# 连续开火多少秒后，这个随机偏差衰减到 0（越打越准）
bot_aim_settle_seconds = 2.0
