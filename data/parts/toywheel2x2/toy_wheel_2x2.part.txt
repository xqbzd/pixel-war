# 玩具轮胎（2x2）：轮轴本体 + 轮体（轮胎）子模型，画在**同一个** 2x2玩具轮胎.vox 里。
#
# 一个零件 = 一个建模文件（2026-10-15 起的约定）：
#   子模型 0 = 轮轴（32 x 19 x 16 体素 = 2 x 2 x 1 格）
#   子模型 1 = 轮体（15 x 32 x 32 体素，外径 2 格）
# 两个物体的**相对位置由画布说了算**：轮轴在画布 y 13..32 / z 16..32，
# 轮体在 x 31..46 / y 0..32 / z 8..40 —— 于是轮体正好挂在外侧那一格、轮心比轮轴中心低 6.5 体素
# （= 原来那种"两个文件 + 手写 wheel_offset"调出来的同一个位置，现在不用手写了）。

name = 玩具轮胎（2x2） / Toy Wheel 2x2

category  = 行驶
model     = res://data/parts/toywheel2x2/2x2玩具轮胎.vox
cubes     = 2 2 1
render    = voxel
order     = 60          # 零件顺序：排在结构件之后、机枪之前
# 连接点：装到车上的那一面是 -x（轮子朝 +x 往外挂，轮胎就在那边）
connections = -x +y +z -z
# 轮体 = 同一个 .vox 里的第 1 号子模型（0 = 轮轴）
wheel_body_model = 1
# 模型偏移的单位为体素，使用游戏内 Y-up 坐标（轮轴整体下移 3 像素，轮体跟着一起动）
model_offset = 0 -3 0
weight = 100
hp = 4000
pp = 20

# 动力和转向。速度使用真实 m/s；内部标定 1 格 = 0.5 米。
wheel_engine_force = 16000       # 每个驱动轮的推力上限，格世界的力
wheel_max_speed = 25             # m/s；推力接近限速时按余弦衰减至 0
wheel_steer_max = 20             # 度
wheel_steer_speed = 60           # 度/秒
wheel_steer_return_speed = 240    # 松手回正速度，度/秒
wheel_radius = 0                 # 0 = 按轮胎模型自动算

# 悬挂：仅需调这两项。每台车进场时读取初始质量 M、轮胎总数 N 和重力 g。
# 单轮分摊质量 m = M / N；刚度 k = 倍率 × m × g / 行程；临界阻尼 c = 2√(k × m)。
# 默认全部车轮触底时，弹簧弹力总和 = 8 × 初始整车重量（阻尼不计入这个定义）。
# 压缩/回弹使用同一临界阻尼，支撑力只推不拉；总支撑力限幅到该轮触底弹簧力。
# 自由长度 = 最大行程，触底长度 = 0，静态压缩量 = 行程 / 倍率。
# 初始硬点补偿静态压缩量，让轮胎保持建造时的高度。
# 这里的静态压缩量不含 data/handling.txt 的额外下压力；下压力会让车身进一步下沉。
# 毁伤会改变刚体重量、重心及存活轮数，但不会重新计算幸存车轮的悬挂参数。
wheel_suspension_travel = 0.6     # 最大避震行程，格（1.2 格 = 0.6 米；最小 0.05）
wheel_support_multiple = 12       # 触底支撑力倍率；默认 8，最小 1
# 原有 stiffness / damper / rest_length / max_force / peak_force 等独立悬挂键已弃用。

# 多射线接触：主射线控制悬挂，整圈射线探测台阶、路缘、坡脚及上方障碍。
# 多个命中合并成一个接触，每个轮子只计算一次弹簧/阻尼/轮胎力。
wheel_rays = 16                  # 整圈采样数；0 = 只保留主射线，方便对比
wheel_ray_slack = 0.04           # 探测余量（格），未真正接触不会施力
wheel_sus_sector = 0.5           # dir·down > 0.5 的下方扇区交给主射线
wheel_ray_offset = 0.4           # 射线沿轮轴向外偏移（格）；不改变视觉轮心

# 侧向沿用 nani_raycast_wheel 的 Pacejka 轮胎曲线（MIT，见 reference/）。
# 纵向直接给推力/刹车力，受支撑力 × long_peak × grip_factor 限制。
wheel_grip_factor = 1.4
wheel_lat_pacejka = 1 10 0 1.35
wheel_long_pacejka = 1 10 0 1.65

# 整车侧倾辅助：取第一个轮轴的数值，与上述车轮弹簧参数分开。
wheel_anti_roll_tilt = 23         # 横向 1 g 时允许倾角（度），0 = 关闭
wheel_anti_roll_damping = 0.55    # 侧倾辅助的临界阻尼倍率
