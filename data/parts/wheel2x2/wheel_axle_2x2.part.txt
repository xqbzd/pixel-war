# ── 轮子参数（只有这个部件用得上；每个轮轴各自读自己这一组）──────────────
# 🚩 2026-10-15（第二次改）：**横向**用 Pacejka（移植自 nani_raycast_wheel.gd，MIT，
#    见 reference/nani_raycast_wheel/）；**纵向**改成"直接给力"（原生 VehicleWheel3D 那套）。
#    上一版把纵向也照它做成"驱动轮力矩 + 轮子自转"，那套在这个尺度下会崩 ——
#    驱动轮力矩远大于轮胎能提供的反作用力矩，轮子会无限自转、推力被摩擦上限顶死、
#    和油门完全脱钩，症状就是"极速设 30 却能开到 60、松油门还在加速"。下面才是现在生效的口径：
#
#   【悬挂】力 = 压缩量 × wheel_stiffness + wheel_damper_compression × 压缩速度
#     wheel_stiffness            弹簧刚度 N/格
#        ⚠️ 实际静止下沉 ≈ 单轮负重 ÷ 刚度 × (1.5 ~ 2.5)：阻尼走隐式积分（不然这个量级的
#           阻尼数值上会炸），稳定但会"稀释"弹簧；稀释倍率随单轮质量变（大车小、小车大）。
#           要"看着压多少"就按实测反推，别按裸刚度算。
#     wheel_damper_compression   阻尼 N·秒/格（**只有这一个生效**；
#                                wheel_damper_relaxation 已不用 —— 新模型只有一份阻尼）
#     wheel_rest_length          悬挂自由长度（格）
#     wheel_suspension_travel    行程上限（格），压到底就顶住
#     wheel_peak_force_car_weights  单轮力上限 = 整车重量 × 它（削尖峰用；0 = 不截断）
#
#   【轮胎】法向载荷 = 这个轮子这一帧的支撑力；轮胎能给的最大力 = 载荷 × peak × grip_factor
#     wheel_lat_pacejka    侧向四系数  peak stiff curve shape（原作者值 1 10 0 1.35）
#       侧向力 = 载荷 × peak × sin(shape·atan(stiff·s − curve·(stiff·s − atan(stiff·s))))
#       s = 速度方向与车轮朝向夹角的**正弦**；小滑移时侧偏刚度 ≈ 载荷 × peak × stiff × shape
#       想"更咬地"就加 peak 或 stiff。
#     wheel_long_pacejka   纵向四系数 —— **只用到第 1 个 peak 当"摩擦上限"**（和横向同口径）
#     wheel_grip_factor    抓地总倍率：>1 更粘，<1 更滑（侧向力和上下限一起缩放）
#     ★ 低速另有一道"防抖闸"：轮胎力不超过"一步之内把这个轮子的滑移速度削平"所需的力。
#       Pacejka 是按滑移**角**算力的，静止/低速时那个角被数值噪声推到 ±90° 直接饱和，
#       力就会帧间来回顶满（车原地抖、横向力 ±19 kN 来回）。加上这道闸之后实测：
#       静置 4/4 着地、速度 0.00、单轮支撑力正好等于静重。
#
#   【驱动 / 刹车】纵向是**直接给车身的力**（不是力矩；轮子已经没有自转状态了）：
#     wheel_engine_force   每个驱动轮的推力上限 N
#       ⚠️ 真正落到地上的力 = min(它 × 限速衰减, 载荷 × peak × grip_factor)。
#          也就是说**抓地上限比电机小的时候，加速由抓地决定、这个键调不出效果** ——
#          想用推力调加速，就把它压到抓地上限以下（现在这台的抓地上限 ≈ 载荷 × 1.0）。
#     wheel_max_speed      限速 m/s：推力按 cos(π/2 × v / vmax) 衰减，**到 vmax 时正好为 0**
#       ⚠️ 实际极速会比它低一点（被轮胎侧偏阻力 + Godot 的 linear_damp 吃掉 5~10%）
#          —— 实测：写 30 → 小车 28、大车 26.4。想要 30 就写 33 左右。
#     刹车：按反方向键时给 -前进速度 × wheel_brake_gain，同样按摩擦上限截断；
#           没油门又几乎不动时也走这一路（低速驻车，不然松手后车会一直"漂"着走）。
#
#   【转向 / 视觉】照旧生效：
#     wheel_steer_max / wheel_steer_speed / wheel_steer_return_speed
#     wheel_rays / wheel_ray_slack / wheel_sus_sector / wheel_ray_offset
#
#     wheel_grip_gain、wheel_slip_falloff、wheel_slip_ref、wheel_brake、
#     wheel_brake_gain、wheel_roll_gain、wheel_damper_relaxation、wheel_mu_per_friction、
#     wheel_mu_max、wheel_mass。
# 标定（2026-10-14 改）：1 cube = 0.5 米 ⇒ project.godot 的重力是 19.6 cube/s²
# （= 真实 9.8 m/s²）。**下面这些"力"不跟着标定变** —— 它们是"格世界"里的数，
# 决定的是 a/g（动力/重力比）这个手感；标定变小/变大时它们要不要缩，是另一回事。
# ⚠️ 标定从 0.2 → 0.5 之后，重力相对几何弱了 2.5 倍：同样的力 + 同样的限速下，
#    车会显得"更冲"（起步更容易抬头、能跳更远），速度相对车身也慢下来了。
#    想找回原来的 a/g：把这里的力（engine/brake/stiffness/damper/max_force）都 ÷ 2.5。
#
# ★ 手感怎么调（2026-10-12 整理，前面的"像在冰面上开车"就是这么来的）：
#   1.8 就明显"粘地"。这个值现在**直接写在下面 `wheel_mu`**（不再从 wheel_friction 反算）。
#   老版本 `wheel_friction` 是"大数字 → ×0.06 → 夹到 1.2"：写 20 以上**一点用都没有**
#   （5000 × 0.06 = 300，被夹成 1.2），所以你会觉得"调了没反应"。现在 ：
#     wheel_mu_per_friction，上限 wheel_mu_max。
#   第二重要的就是"跑起来就滑"的那个衰减：**只作用在横向**（`wheel_slip_falloff`）——
#   横向滑得快才掉抓地；直线/刹车用满 mu。调大 = 更容易推头/甩尾，调 0 = 抓地拉满。
#   剩下的顺序建议：先 wheel_mu 定"拐多急"，再 wheel_stiffness / wheel_damper_*（悬挂软硬），
#   最后 wheel_anti_roll_tilt（防翻多硬）和 wheel_engine_force（动力）找手感。

# 动力
wheel_engine_force        = 15000   # 每个驱动轮的推力，N
# ⚠️ 速度单位统一成**真实 m/s**（1 cube = 0.5 米）。以前这里标着 m/s、实际是格/秒：
#    150 格/秒 = 30 m/s，所以现在写 30。（内部换算 ×1/CUBE_METERS = ×2 只做一次）
wheel_max_speed           = 30     # 超过这个速度就收油，m/s
wheel_steer_max           = 15    # 最大转角，度
wheel_steer_speed         = 60    # 转向角速度，度/秒（越大越跟手，越小越"缓启动"；0 会被夹到 1）
# ★ 松手回正角速度，度/秒：松开方向键后转角回中位的速度。**单独一条**，不影响上面
#   打方向那段（想"撒手就正"就写 240~360；不写 = 和 wheel_steer_speed 同速 = 老行为）。
wheel_steer_return_speed  = 240
# 半径：0 = 按轮胎模型自动算（轮胎盒子 y/z 较大者的一半）
wheel_radius              = 0

# ── 抓地力（★ 最要紧的两行）──────────────────────────────────────────────

# ── 整圈射线的形状 ────────────────────────────────────────────────────────
# wheel_rays = 0 就是**纯单射线模式**（整圈射线全关，只剩悬挂那条射线）——
# 想要"最早那种单射线手感"、或者想确认问题出在底盘还是出在整圈射线，直接写 0 对比。
wheel_rays                = 16    # 每圈打几条（360/rays 度一条）；少一点省性能
wheel_ray_slack           = 0.04  # 射线长度 = 半径 + 这个余量（提前一点发现台阶边缘）
# 与"下"方向夹角小于 60°（点积 > 0.5）的射线**跳过**：轮子正下方那片归悬挂那条射线管。
# ⚠️ 别调成 0 以下：那会让正下方的射线也参与，等于把弹簧/阻尼/抓地并联 2~3 份，
# 而且随压缩量跳变 —— 症状就是"平地也飘、还不稳"。之前判据写反（用"上"判）踩过这个坑。
wheel_sus_sector          = 0.5
# 判定射线的**侧向偏移**（格，沿轮胎轴向，正 = 往车外）：
#   0    = 射线打在**轮胎中缝**那个平面上（老行为）
#   0.4  = 挪到**轮胎最外侧面**（这张轮胎 0.81 格宽，一半就是 0.4）★ 现在这个
# 挪的是射线本身：轮胎网格、轮心、悬挂高度都不变，只是"贴路缘/蹭墙时咬合的位置"从
# 轮胎中缝变成外侧面。写太大会让判定跑到轮胎外面（看着像悬空咬合），一般别超半宽。
wheel_ray_offset          = 0.4

# ── 力的增益（目标力 = 相对速度 × 增益，再按 mu × 法向力截断）────────────

# ── 悬挂（★ 全是可直接填的物理量，没有倍率/换算）──────────────────────────
#
# 单位：长度 = **格**（1 格 = 0.5 米）；力 = 格世界的 N；阻尼 = N·秒/格。
# 这些是"格世界"里的数（重力 19.6 格/秒² = 真实 9.8 ÷ 0.5），别拿真实米/秒去算。
#
# 只有三个数决定悬挂手感：
#   ① wheel_stiffness          弹簧多硬（N/格）= 压 1 格需要多少力
#   ② wheel_damper_*           阻尼（N·秒/格）= 吃不吃振荡；压缩/回弹分开调
#   ③ wheel_suspension_travel  最多能压多少格（压到底就顶死，防止穿透地面）
#
# 静止时压下去多少**是算出来的**，不用你填：
#     静止压缩量 = (整车重量 ÷ 轮子数) ÷ wheel_stiffness
#   例（608 kg 那台）：608 ÷ 8 轮 = 76 kg/轮 → 重量 = 76 × 19.6 = 1490 N
#     · 想要静止压 0.06 格（软一点）：1490 ÷ 0.06 ≈ **25000**
#     · 写 2480000 → 只压 0.0006 格，等于几乎压不动：车硬、颠、侧倾小
#   底盘高度不用管：硬点是按"静止压缩量"反推的，轮胎底面永远落在你摆的位置上。
#
# 阻尼怎么选：临界阻尼 c = 2 × √(刚度 × 单轮质量)，取它的 0.5~1.5 倍一般都舒服。
#   你这套（530 kg 小车）：生效刚度 ≈ 132000、单角簧上质量 133 kg → 临界 ≈ 8380；
#   写 20000 经隐式积分折成 ≈ 8900 ≈ 1.06 倍临界（正好不过冲、也不发闷）。
#   比临界小 = 过坎弹两下；比临界大很多 = 发闷。
# ⚠️ 下面这四项是**配套的**，改一个要跟着改另外几个（不然会出现"车贴地走/轮胎入地"）。
# 悬挂参数是**零件固有值**（绝对值，和车重无关）。曾经试过"按车重自动算"那套，撤了 ——
# 轻车被弹飞的根因不是比值，而是"射线式悬挂会在临界高度给出**一帧跳变**的力"，
# 那个用**低通滤波**治（`SUSP_FORCE_FILTER_SECONDS`，见 scripts/test_field.gd）。
#   实测（2026-10-15，530 kg 小车 / 133 kg 单角簧上质量）：这个 300000 的刚度 +
#   下面 20000 的阻尼，静止压缩量是 **0.0295 格**（= 单轮静重 3895 N ÷ 实际生效刚度 132000）。
wheel_stiffness            = 300000    # 弹簧刚度 N/格
wheel_damper_compression   = 2000     # 压缩时的阻尼 N·秒/格（隐式积分，实际生效值 ≈ 1/2.25）
# ★ 行程上限（格）：**至少要是"静止压缩量"的 5 倍**，软悬挂建议 15~20 倍。
#   实测静止压 0.0295 格 → 写 1.2 = 40 倍（60 厘米行程，过坎、落地都够用）。
#   写 0.3（= 5 倍、6 厘米）会**经常到底**：到底之后支撑力停在 k×行程 上不再涨，
#   车身就贴着地面走、轮胎看着像陷进地里（实测 8 个轮子里 4 个卡在 0.3 上限）。
wheel_suspension_travel    = 1.2
wheel_rest_length          = 0.15      # 悬挂自由长度，格：硬点比轮心高这么多
# 支撑力上限 N：安全阀，正常行驶碰不到。⚠️ 刚度调得很硬时它会开始起作用（过坎像撞硬限位），
# 那一档要把它一起放大（比如 100000），否则"调硬"会被这个上限吃掉。
wheel_suspension_max_force = 150000    # 上限要高于"刚度 × 行程"（62000 × 1.2 = 74400），否则"调软"会被它吃掉

# ── 侧倾稳定器（防翻）：整车共用一份，取**第一个轮轴**的这两个值 ──────────
# 它做的事：有轮子着地时，给车身一个"回正弹簧 + 阻尼"，不让它在转弯时越倾越厉害。
#   wheel_anti_roll_tilt  = 横向 1 g 时最多允许倾多少**度**（★ 单位是度，以前是弧度）
#                           调小 = 更硬（更不容易翻，但会像"有根看不见的杆子"）；写 0 = 关掉
#   wheel_anti_roll_damping = 阻尼相对临界阻尼的倍率（1.0 = 刚好不来回摆）
wheel_anti_roll_tilt      = 23   # 度（≈ 横向 1 g 时倾 23°；0 = 关掉防翻）
wheel_anti_roll_damping   = 0.55
#   单轮支撑力的**硬截断**（用户手调）：一个轮子的力 ≤ 整车重量 × 它。1 = 一倍车重；0 = 不截断。
#   射线式悬挂在临界高度上会给出"一帧满力"的尖峰，截断能削掉它而且**不改相位**
#   （低通滤波会加滞后、把支撑回路推成自激，连两吨的车都抖 —— 试过，别用）。
#   静止时单轮只有"车重 ÷ 轮数"，离上限很远，所以只削尖峰、不影响手感。
wheel_peak_force_car_weights = 4

# 已失效（纵向改成"直接给力"之后就没有轮子自转状态了）：wheel_mass。
# ── 轮胎力模型（Pacejka，移植自 nani_raycast_wheel.gd，MIT，见 reference/ 目录）──
#   **侧向**力 = 法向载荷 × peak × sin(shape·atan(stiff·滑移 − curve·(stiff·滑移 − atan(stiff·滑移))))
#   小滑移时 ≈ 载荷 × peak × stiff × shape（侧偏刚度），大滑移自动饱和（不会把车掀翻）。
#   纵向只取第 1 个 peak 当摩擦上限，推力本身用 wheel_engine_force（见上面【驱动/刹车】）。
#   下面三个键的默认值就是原作者那份，先按原样跑。
wheel_grip_factor          = 1              # 抓地总倍率：>1 更粘地，<1 更滑
wheel_lat_pacejka          = 1 10 0 1.35    # 侧向四系数：peak stiff curve shape
wheel_long_pacejka         = 1 10 0 1.65    # 纵向四系数：peak stiff curve shape
