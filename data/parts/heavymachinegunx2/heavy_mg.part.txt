# 重机枪（5x5x5）—— 轻机枪的重型版：同一套转塔代码，模型换成 重机枪2.vox。
#
# 数值目前**完全照抄轻机枪**（伤害 / 射速 / 散射 / 能量 …），等着手填。
# 唯一不同的是枪声：`weapon_pitch = 0.78` → 比轻机枪低沉一档。
#
# 模型 `重机枪2.vox` 是 MagicaVoxel 场景（80×80×80 体素 = 5×5×5 cube），里面三个物体：
#   0 = 底座（固定）、1 = 支架（水平 yaw）、2 = 枪管（俯仰 pitch）
# 转轴 / 转点 / 枪口都从模型几何推导（PartDef.weapon_*），换模型不用改这里。
#
name = 重机枪 / Heavy MG

category   = 武器
model      = res://data/parts/heavymachinegunx2/重机枪2.vox
cubes      = 5 5 5
render     = voxel
order      = 67          # 排在双联装机枪之后、核心之前
# 连接点：只有底面有，而且**只有基准格那一格**（1×1 的小底板 → 正好 1 格）
connections = -y

# ── 武器：转塔 ─────────────────────────────────────────────────────────────
weapon = turret
# 枪声音高（1.0 = 轻机枪原样）：重机枪更低沉
weapon_pitch = 0.78
weapon_yaw_model   = 1    # 支架：相对底座水平旋转（子模型序号）
weapon_pitch_model = 2    # 枪管：相对支架上下俯仰
# 水平射界（整角，度）：360 = 全向自由旋转；< 360 就以枪口正前方为中心 ±一半。
# 角度一律走最短路径（跨 ±180° 也是 20° 就是 20°，不会绕远转 340°）。
weapon_yaw_limit   = 360
# 俯仰范围（度，相对水平）：负 = 压枪，正 = 抬枪
weapon_pitch_min   = -20
weapon_pitch_max   = 60

# ── 开火 ───────────────────────────────────────────────────────────────────
# ⚠️ 速度一律写**真实 m/s**（世界标定 1 cube = 0.5 米；以前车轮那个"m/s"其实是格/秒，
# 已经统一过来了，测试场内部换算成格）。下面这些数先随便填，等你手调。
weapon_projectile   = ray     # 射线类：瞬间命中，没有弹速/下坠/寿命，只有最大射程
weapon_ray_range    = 400     # 最大射程，米（射线类专用）
weapon_muzzle_speed = 800     # 子弹初速度，m/s
weapon_damage       = 400     # 每发伤害
weapon_rpm          = 300     # 射速，发/分钟
weapon_spread       = 0.18     # 初速度的随机角散射，度（±，锥形）
weapon_traverse_speed = 660    # 炮塔最高转速，度/秒（水平、俯仰共用这一个）
weapon_energy_cost  = 100      # 每发耗能（核心提供 10000 上限 + 1000/s 恢复）
# 弹丸寿命（秒）：打出去多久自动消失。**射速 × 寿命 = 同屏最多几发**，防卡爆主要靠它
# （另一道保险是射弹池硬上限 256 发）。写 0 会被夹成 0.05 秒，上限 120 秒。
weapon_projectile_life = 0.5

# ── 弹头（普通子弹）────────────────────────────────────────────────────────
# 这四个是"打出去的这一发"自己的属性，默认值 = 加这几个参数之前的行为（手感不变）。
# 后坐冲量：每发把**自己这台车**往后推多少。冲量 = 质量 × 速度，单位 kg·格/秒
# （1 格 = 0.5 米），作用点在枪口，所以偏心装的枪会带一点力矩。
#   例：2000 kg 的车写 100 → 每发 0.05 格/秒（0.025 m/s）后退；0 = 没后坐。
weapon_recoil_impulse = 100
# 支架（水平旋转部分）要不要独立碰撞盒：体积小的枪用 false（并进车体合并盒，省一个 shape）；
# 大炮塔再打开（以后会做成"跟着旋转的碰撞盒"）
weapon_swivel_collider = false
# 冲击冲量：每发打中**别人**时把对方推多少（同上单位，作用点在命中点）。
# 打到地形/静态体推不动（只对刚体生效）。0 = 不推。
weapon_impact_impulse = 50
# 弹头颜色（十六进制 RRGGBB）：**曳光轨迹 + 弹着点火花**都用它。默认 = 现在的黄色。
weapon_bullet_color = ffd959
# 距离衰减：每飞 100 米损失多少伤害比例（0.1 = 每百米 -10%；1 = 百米外没伤害）。
# 0 = 不衰减。距离按真实米数算（1 格 = 0.5 米），所以 0.1 等于"每 200 格 -10%"。
weapon_damage_falloff = 0.1

# 基本数值（先按默认填，等手动调）
weight    = 15
hp        = 10000
pp        = 30
